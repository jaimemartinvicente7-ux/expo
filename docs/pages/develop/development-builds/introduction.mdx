import React, { useEffect, useState, useRef } from "react";
import { View, Text, TouchableOpacity, StyleSheet, ScrollView, Alert, Platform } from "react-native";
import { Camera } from "expo-camera";
import * as Location from "expo-location";
import * as Speech from "expo-speech";
import AsyncStorage from "@react-native-async-storage/async-storage";
import axios from "axios";

export default function App() {
  const [hasCameraPermission, setHasCameraPermission] = useState(false);
  const [hasLocationPermission, setHasLocationPermission] = useState(false);
  const [memory, setMemory] = useState([]);
  const [location, setLocation] = useState(null);
  const cameraRef = useRef(null);

  useEffect(() => {
    (async () => {
      // Pedir permisos
      const { status: cameraStatus } = await Camera.requestCameraPermissionsAsync();
      setHasCameraPermission(cameraStatus === "granted");

      const { status: locStatus } = await Location.requestForegroundPermissionsAsync();
      setHasLocationPermission(locStatus === "granted");

      await loadMemory();
    })();
  }, []);

  const loadMemory = async () => {
    try {
      const savedMemory = await AsyncStorage.getItem("elena_memory");
      if (savedMemory) setMemory(JSON.parse(savedMemory));
    } catch (e) {
      console.log("Error cargando memoria:", e);
    }
  };

  const saveMemory = async (msg) => {
    try {
      const newMemory = [...memory, msg];
      setMemory(newMemory);
      await AsyncStorage.setItem("elena_memory", JSON.stringify(newMemory));
    } catch (e) {
      console.log("Error guardando memoria:", e);
    }
  };

  const speak = (text) => {
    Speech.speak(text, {
      voice: Platform.OS === "ios" ? "com.apple.ttsbundle.siri_female_en-US_compact" : undefined,
      rate: 0.9,
    });
  };

  const handleQuery = async (query) => {
    saveMemory({ type: "query", content: query, timestamp: Date.now() });

    if (query.toLowerCase().includes("elena, siguiente nivel")) {
      speak("Modo dios activado");
      return;
    }

    try {
      const response = await axios.post("https://TU_BACKEND", { query, memory });
      const answer = response.data.answer || "No entendí eso.";
      saveMemory({ type: "answer", content: answer, timestamp: Date.now() });
      speak(answer);
    } catch (e) {
      console.log("Error backend:", e);
      speak("Error de conexión al servidor.");
    }
  };

  const takePicture = async () => {
    if (!cameraRef.current) return speak("Cámara no disponible.");
    try {
      const photo = await cameraRef.current.takePictureAsync();
      saveMemory({ type: "photo", uri: photo.uri, timestamp: Date.now() });
      speak("Foto tomada.");
    } catch {
      speak("Error al tomar la foto.");
    }
  };

  const getLocation = async () => {
    if (!hasLocationPermission) return speak("No tengo permiso de ubicación.");
    try {
      const loc = await Location.getCurrentPositionAsync({});
      setLocation(loc);
      saveMemory({ type: "location", coords: loc.coords, timestamp: Date.now() });
      speak(`Latitud ${loc.coords.latitude}, longitud ${loc.coords.longitude}`);
    } catch {
      speak("Error obteniendo ubicación.");
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Elena</Text>

      {hasCameraPermission ? (
        <Camera ref={cameraRef} style={styles.camera} />
      ) : (
        <Text style={{ color: "#fff" }}>No hay permiso de cámara</Text>
      )}

      <View style={styles.controls}>
        <TouchableOpacity style={styles.button} onPress={takePicture}>
          <Text style={styles.buttonText}>Tomar Foto</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.button} onPress={getLocation}>
          <Text style={styles.buttonText}>Ubicación</Text>
        </TouchableOpacity>

        <TouchableOpacity style={styles.button} onPress={() => handleQuery("Elena, siguiente nivel")}>
          <Text style={styles.buttonText}>Siguiente Nivel</Text>
        </TouchableOpacity>
      </View>

      <ScrollView style={styles.memory}>
        <Text style={styles.memoryTitle}>Memoria:</Text>
        {memory.map((m, i) => (
          <Text key={i} style={styles.memoryText}>
            {m.type === "query" ? `Q: ${m.content}` : m.type === "answer" ? `A: ${m.content}` : m.type === "photo" ? `Foto: ${m.uri}` : m.type === "location" ? `Loc: ${JSON.stringify(m.coords)}` : JSON.stringify(m)}
          </Text>
        ))}
      </ScrollView>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: "#111", paddingTop: 50 },
  title: { fontSize: 24, color: "#0ff", textAlign: "center", marginBottom: 10 },
  camera: { flex: 1, margin: 10, borderRadius: 20, overflow: "hidden" },
  controls: { flexDirection: "row", justifyContent: "space-around", marginVertical: 10 },
  button: { backgroundColor: "#0ff", padding: 10, borderRadius: 10 },
  buttonText: { color: "#111", fontWeight: "bold" },
  memory: { flex: 1, margin: 10 },
  memoryTitle: { color: "#0ff", fontWeight: "bold", marginBottom: 5 },
  memoryText: { color: "#fff", fontSize: 12 },
});
